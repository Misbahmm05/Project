
import json
import boto3

dynamodb = boto3.resource('dynamodb')
table = dynamodb.Table('StudentRegistration')

def lambda_handler(event, context):
    data = json.loads(event['body'])

    name = data['name']
    rollno = data['rollno']
    branch = data['branch']

    table.put_item(
        Item={
            'rollno': rollno,
            'name': name,
            'branch': branch
        }
    )

    return {
        'statusCode': 200,
        'headers': {
            'Content-Type': 'application/json'
        },
        'body': json.dumps({
            'message': 'Registration successful',
            'name': name,
            'rollno': rollno,
            'branch': branch
        })
    }

<!DOCTYPE html>
<html>
<head>
    <title>Student Registration</title>
</head>
<body>

    <h2>Student Registration Form</h2>

    <form id="registrationForm">
        <label>Name:</label>
        <input type="text" id="name" required>
        <br><br>

        <label>Roll Number:</label>
        <input type="text" id="rollno" required>
        <br><br>

        <label>Branch:</label>
        <input type="text" id="branch" required>
        <br><br>

        <button type="submit">Register</button>
    </form>

    <p id="result"></p>

    <script>
        document.getElementById("registrationForm")
        .addEventListener("submit", async function(event) {
            event.preventDefault();

            const studentData = {
                name: document.getElementById("name").value,
                rollno: document.getElementById("rollno").value,
                branch: document.getElementById("branch").value
            };

            try {
                const response = await fetch(
                    "https://vuo3b7t9y5.execute-api.us-east-1.amazonaws.com/register",
                    {
                        method: "POST",
                        headers: {
                            "Content-Type": "application/json"
                        },
                        body: JSON.stringify(studentData)
                    }
                );

                const result = await response.json();

                document.getElementById("result").textContent =
                    response.ok
                    ? result.message
                    : "Registration failed: " + (result.message || "Please try again.");

            } catch (error) {
                document.getElementById("result").textContent =
                    "Error connecting to the server. Check API Gateway and CORS.";
            }
        });
    </script>

</body>
</html>
